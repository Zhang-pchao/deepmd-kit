cmake_minimum_required(VERSION 3.25.2)
project(selected_graph_tests LANGUAGES CXX)
find_package(Kokkos CONFIG REQUIRED)
find_package(GTest CONFIG QUIET)
if(NOT GTest_FOUND)
  if(NOT IS_DIRECTORY "${GTEST_SOURCE_DIR}")
    message(FATAL_ERROR "Provide GTest_DIR or an existing GTEST_SOURCE_DIR; no automatic downloads")
  endif()
  add_subdirectory("${GTEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/gtest" EXCLUDE_FROM_ALL)
endif()
add_executable(selected_graph_test test_selected_graph.cpp test_selected_model.cpp)
target_compile_features(selected_graph_test PRIVATE cxx_std_17)
target_include_directories(selected_graph_test PRIVATE "../../../../api_c/include" "../../../../api_cc/include")
target_link_libraries(selected_graph_test PRIVATE Kokkos::kokkos GTest::gtest)
enable_testing()
add_test(NAME selected_graph COMMAND selected_graph_test)
